int contadorGlobal;
int numeros[3];

int sumar(int a, int b) {
    int resultado;
    resultado = a + b;
    return resultado;
}

int main() {
    int i;
    int total;

    i = 0;
    total = 0;

    numeros[0] = 10;
    numeros[1] = 20;
    numeros[2] = 30;

    total = sumar(numeros[0], numeros[1]);

    while (i < 3) {
        total = total + numeros[i];
        i = i + 1;
    }

    for (int j = 0; j < 3; j = j + 1) {
        if (j == 2) {
            continue;
        }

        total = total + j;
    }

    if (total > 0) {
        contadorGlobal = total;
    }
    else {
        contadorGlobal = 0;
    }

    return contadorGlobal;

    contadorGlobal = 999;
}
