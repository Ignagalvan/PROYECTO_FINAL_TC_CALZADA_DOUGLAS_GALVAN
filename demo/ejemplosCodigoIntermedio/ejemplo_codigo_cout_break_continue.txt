int x = 0;
int suma = 0;

cout << "Inicio" << x;

while (x < 5) {
    x = x + 1;
    cout << "x vale" << x;

    if (x == 2) {
        continue;
    }

    if (x == 4) {
        break;
    }

    suma = suma + x;
}

for (int i = 0; i < 3; i = i + 1) {
    cout << "i vale" << i;

    if (i == 1) {
        continue;
    }

    suma = suma + i;
}

cout << "Fin" << suma;
